import { notFound } from "next/navigation";
import StandingsTable from "@/components/StandingsTable";
import MatchList from "@/components/MatchList";
import ScorersTable from "@/components/ScorersTable";
import { getDict } from "@/lib/dictionary";
import { getLeague } from "@/lib/leagues";
import { getStandings, getMatches, getScorers } from "@/lib/api";

export const revalidate = 300;

export async function generateMetadata({ params }) {
  const { lang, code } = await params;
  const league = getLeague(code);
  if (!league) return {};
  const t = getDict(lang);
  return { title: `${league.name[lang]}: ${t.standings}` };
}

export default async function LeaguePage({ params }) {
  const { lang, code } = await params;
  const league = getLeague(code);
  if (!league) notFound();
  const t = getDict(lang);

  const [standings, matches, scorers] = await Promise.all([
    getStandings(league.code),
    getMatches(league.code),
    getScorers(league.code),
  ]);
  const demo = standings.demo || matches.demo || scorers.demo;
  const error = standings.error || matches.error || scorers.error;

  return (
    <>
      <h1>
        {league.flag} {league.name[lang]}
      </h1>
      {demo && <div className="notice">{t.demoNotice}</div>}
      {error && <div className="notice">{t.errorNotice}</div>}

      <h2>{t.standings}</h2>
      <StandingsTable table={standings.table} lang={lang} />

      <h2>{t.matches}</h2>
      <MatchList matches={matches.matches} lang={lang} />

      <h2>{t.scorers}</h2>
      <ScorersTable scorers={scorers.scorers} lang={lang} />
    </>
  );
}

"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";

// Переключатель языка: заменяет первый сегмент пути (/ru/... <-> /en/...)
export default function LangSwitch({ lang }) {
  const pathname = usePathname() || `/${lang}`;
  const rest = pathname.split("/").slice(2).join("/");
  const build = (l) => `/${l}${rest ? "/" + rest : ""}`;

  return (
    <div className="lang">
      {["ru", "en"].map((l) => (
        <Link
          key={l}
          href={build(l)}
          className={l === lang ? "active" : ""}
        >
          {l.toUpperCase()}
        </Link>
      ))}
    </div>
  );
}

import { getDict } from "@/lib/dictionary";

const LIVE = ["IN_PLAY", "PAUSED"];

function formatDate(iso, lang) {
  return new Date(iso).toLocaleString(lang === "ru" ? "ru-RU" : "en-GB", {
    day: "2-digit",
    month: "short",
    hour: "2-digit",
    minute: "2-digit",
    timeZone: "Europe/Minsk", // часовой пояс: поменяйте на свой, например "Europe/Moscow"
  });
}

function Crest({ team }) {
  if (!team.crest) return null;
  return <img src={team.crest} alt="" width="20" height="20" loading="lazy" />;
}

export default function MatchList({ matches, lang, emptyText }) {
  const t = getDict(lang);
  if (!matches.length) return <p className="muted">{emptyText ?? t.noData}</p>;

  const sorted = [...matches].sort(
    (a, b) => new Date(a.utcDate) - new Date(b.utcDate)
  );

  return (
    <ul className="matches">
      {sorted.map((m) => {
        const { home, away } = m.score?.fullTime ?? {};
        const hasScore = home !== null && home !== undefined;
        const live = LIVE.includes(m.status);
        return (
          <li key={m.id} className="match">
            <span className="match-date">
              {live ? <b className="live">{t.statusLive}</b> : formatDate(m.utcDate, lang)}
            </span>
            <span className="match-team right">
              <span>{m.homeTeam.shortName || m.homeTeam.name}</span>
              <Crest team={m.homeTeam} />
            </span>
            <span className="match-score">{hasScore ? `${home} : ${away}` : "–"}</span>
            <span className="match-team">
              <Crest team={m.awayTeam} />
              <span>{m.awayTeam.shortName || m.awayTeam.name}</span>
            </span>
          </li>
        );
      })}
    </ul>
  );
}

import { DEMO_TABLE, DEMO_MATCHES, DEMO_SCORERS } from "./demo";
import { LEAGUES } from "./leagues";

const BASE = "https://api.football-data.org/v4";
const KEY = process.env.FOOTBALL_DATA_API_KEY;

// Данные кэшируются на 5 минут, чтобы не превышать бесплатный лимит (10 запросов в минуту).
const REVALIDATE_SECONDS = 300;

// Возвращает данные или null, если запрос не удался (неверный ключ, лимит, сеть).
async function call(path) {
  try {
    const res = await fetch(BASE + path, {
      headers: { "X-Auth-Token": KEY },
      next: { revalidate: REVALIDATE_SECONDS },
    });
    if (!res.ok) return null;
    return await res.json();
  } catch {
    return null;
  }
}

function isoDate(offsetDays = 0) {
  const d = new Date();
  d.setUTCDate(d.getUTCDate() + offsetDays);
  return d.toISOString().slice(0, 10);
}

// Во всех функциях: demo = true, если ключа нет; error = true, если ключ есть, но данные не пришли.

export async function getStandings(code) {
  if (!KEY) return { demo: true, table: DEMO_TABLE };
  const data = await call(`/competitions/${code}/standings`);
  const total = data?.standings?.find((s) => s.type === "TOTAL");
  if (!total) return { error: true, table: [] };
  return { table: total.table };
}

export async function getMatches(code) {
  if (!KEY) return { demo: true, matches: DEMO_MATCHES };
  // Окно: 5 дней назад и 4 дня вперёд (API не любит окна длиннее 10 дней)
  const data = await call(
    `/competitions/${code}/matches?dateFrom=${isoDate(-5)}&dateTo=${isoDate(4)}`
  );
  if (!data) return { error: true, matches: [] };
  return { matches: data.matches };
}

export async function getScorers(code) {
  if (!KEY) return { demo: true, scorers: DEMO_SCORERS };
  const data = await call(`/competitions/${code}/scorers?limit=15`);
  if (!data) return { error: true, scorers: [] };
  return { scorers: data.scorers };
}

export async function getTodayMatches() {
  if (!KEY) return { demo: true, matches: [] };
  const today = isoDate(0);
  const codes = LEAGUES.map((l) => l.code).join(",");
  const data = await call(
    `/matches?competitions=${codes}&dateFrom=${today}&dateTo=${today}`
  );
  if (!data) return { error: true, matches: [] };
  return { matches: data.matches };
}
